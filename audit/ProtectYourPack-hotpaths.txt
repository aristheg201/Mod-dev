   132	    static void deleteRecursively(Path path) throws IOException {
   133	        if (path == null || !Files.exists(path, new LinkOption[0])) {
   134	            return;
   135	        }
   136	        try (Stream<Path> stream = Files.walk(path, new FileVisitOption[0]);){
   137	            for (Path path2 : stream.sorted(Comparator.reverseOrder()).toList()) {
   138	                Files.deleteIfExists(path2);
   139	            }
   140	        }
